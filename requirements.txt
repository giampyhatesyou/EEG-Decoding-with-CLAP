# ===== Core ML stack (pinned for API compatibility) =====
# torch 2.2.2 requires numpy < 2.0
torch==2.2.2
torchaudio==2.2.2
numpy==1.26.4

# pytorch_lightning 1.9.x uses old API (Trainer.from_argparse_args, resume_from_checkpoint kwarg)
# Upgrading to 2.x breaks main.py and checkpoint_test.py
pytorch_lightning==1.9.5

# torchmetrics MUST be pinned: the default resolver pulls torchmetrics 1.9+,
# which imports `torchmetrics.functional.image.arniqa` -> `torchvision`. With
# torch 2.2.2 + torchvision 0.17.x this triggers a circular-import crash
# ("partially initialized module 'torchvision' has no attribute 'extension'")
# the moment Lightning imports torchmetrics. 0.11.x is the version validated
# against Lightning 1.9.x and does not pull torchvision at import time.
torchmetrics==0.11.4

# ===== Data processing =====
pandas==1.5.3
scikit_learn==1.3.2
scipy==1.11.4
PyYAML==6.0.2

# ===== Audio =====
audiomentations==0.27.0

# LAION-CLAP for the CLAP audio-representation experiment (see src/models/clap_encoder.py).
# Optional in spirit — the codebase still trains the original SampleCNN2DEEG baseline
# when `audio_repr: "raw"` in the YAML config. We pin a specific release because
# the laion_clap upstream API has historically been unstable across patch versions.
laion-clap==1.1.6

# laion_clap 1.1.6 imports `BertModel`/`RobertaModel`/`BartModel` directly from
# transformers, which broke in transformers >= 4.40 when those classes moved
# behind lazy module loading. Pin to a version that still re-exports them.
transformers<4.40

# laion_clap declares numpy==1.23.5 as a hard pin, but the codebase actually
# works fine with 1.26.x (the pin is over-strict). We override here because
# torch 2.2.2 + numba 0.65 + scipy 1.11.4 all require numpy >= 1.24; sticking
# to 1.23.5 would break the rest of the stack. The ABI is stable across this
# range, no observed runtime incompatibility.
# numpy==1.26.4 is already pinned above, this comment just records the why.

# ===== Contrastive learning =====
# NT_Xent loss from simclr (used as fallback in EEGContrastiveLearning.configure_criterion)
simclr==1.0.2

# ===== Logging / visualization =====
tensorboard
tqdm==4.66.5

# matplotlib 3.8+ pairs with contourpy 1.3+ which has a rendering bug on this stack:
# 'ValueError: object __array__ method not producing an array' on fig.savefig()
# Pinning to a known-good combination.
matplotlib==3.7.5
contourpy<1.2

# ===== Build tools =====
# setuptools >= 81 removes pkg_resources, breaking pytorch_lightning 1.9.x
setuptools<81
